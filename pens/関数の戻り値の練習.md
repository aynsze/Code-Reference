## 成功バージョン
- ###  1つの関数にする場合
  ```js
  // 関数名「getUserName」を定義
  function getUserName() {
    // 1. 変数名「USER_NAME」では、ユーザー名を定義
    const USER_NAME = "github";

    // 2. 変数名「url」では、(1)からURLリンクを作成する
    const url = `https://github.com/${USER_NAME}`;

    console.log(url);
  }

  // 関数を実行
  getUserName();
  // => https://github.com/github
  ```
  <!-- demo -->

- ### 複数の関数にする場合

  - `getUserName()` から呼び出す
    ```js
    // 関数名「getUserName」を定義
    function getUserName() {
      // 1. 変数名「USER_NAME」では、ユーザー名を定義
      const USER_NAME = "github";

      // (1)を、別の関数「createURL」で使用したいため
      // return にして、(1)を引数として渡す
      return createURL(USER_NAME);
    }

    // 関数名「createURL」を定義して、「getUserName」関数で呼び出した(1)の引数「USER_NAME」を使用する
    function createURL(USER_NAME) {
      // 2. 変数名「url」では、「getUserName」関数で呼び出した(1)の引数「USER_NAME」からURLリンクを作成する
      const url = `https://github.com/${USER_NAME}`;

      console.log(url);
    }

    // 「getUserName」関数を実行（呼び出す）
    getUserName();
    // => https://github.com/github
    ```
    <!-- demo -->

  - `createURL("github")` を直接呼び出す
    ```js
    function getUserName() {
      const USER_NAME = "github";
      return createURL(USER_NAME);
    }

    function createURL(USER_NAME) {
      const url = `https://github.com/${USER_NAME}`;

      console.log(url);
    }

    createURL("github");
    // 直接 "github" を引数に渡しているため、
    // => https://github.com/github
    // しかし、関数「getUserName」は実行されていない
    ```
    <!-- demo -->


## 失敗バージョン
- ### 関数を呼び出していない場合
  ```js
  function getUserName() {
    const USER_NAME = "github";
    return createURL(USER_NAME);
  }

  function createURL(USER_NAME) {
    const url = `https://github.com/${USER_NAME}`;
    console.log(url);
  }

  // => 実行されない
  ```
  <!-- demo -->

- ### `createURL()` を引数なしで呼び出す場合
  ```js
  function getUserName() {
    const USER_NAME = "github";
    return createURL(USER_NAME);
  }

  function createURL(USER_NAME) {
    const url = `https://github.com/${USER_NAME}`;
    console.log(url);
  }

  // 「createURL」関数を実行（呼び出す）
  createURL();
  // 引数「USER_NAME」が渡されていないため、undefinedになる
  // => https://github.com/undefined
  ```
  <!-- demo -->

- ### `createURL(USER_NAME)` を直接呼び出す場合
  ```js
  function getUserName() {
    const USER_NAME = "github";
    return createURL(USER_NAME);
  }

  function createURL(USER_NAME) {
    const url = `https://github.com/${USER_NAME}`;
    console.log(url);
  }

  // 「createURL」関数を実行（呼び出す）
  createURL(USER_NAME);
  // => エラーで実行されない
  ```
  <!-- demo -->

- ### `return url;` を追加した場合
  ```js
  function getUserName() {
    const USER_NAME = "github";
    return createURL(USER_NAME);
  }

  function createURL(USER_NAME) {
    // 2. 変数名「url」では、「getUserName」関数で呼び出した(1)の引数「USER_NAME」からURLリンクを作成する
    const url = `https://github.com/${USER_NAME}`;

    // (2)を、他でも使用したいため
    // return にする
    return url;
  }

  console.log(url);
  // (2) は、関数「createURL」の中で定義した変数なので、関数の外から直接参照できない
  // => エラー
  ```
  <!-- demo -->


## 説明
戻り値の`変数名` とは、その変数を他でも使用したいときに使用

- ### `return 使用したい関数名(変数名)
  ```js
  return createURL(USER_NAME);

  function createURL(USER_NAME){
    ...
  };
  ```

- ### `return 変数名`
  変数名だけを返すため、ほかの関数でも使用可能
  ```js
  return USER_NAME;

  function 関数1(USER_NAME){
    ...
  }

  function 関数2(USER_NAME){
    ...
  }
  ```
