## 2. 倍率に沿った文字を説明に表示

```js
// 1. HTMLのid名「servingNote」を取得
const note = document.getElementById("servingNote");

// 2. HTMLのid名「servingSelect」を取得
const select = document.getElementById("servingSelect");

// ➁の選択内容が変更（change）されたときに実行
select.addEventListener("change", () => {

  // 3. ➁のvalueの値を数値に変換
  const ratio = Number(select.value);

  // 4. switch文を使用した条件分岐
  switch (ratio) {

    // ratio === 2のときの処理
    case 2:
      // ➀に文字列を設定
      note.textContent = "2倍の分量";
      break;

    // ratio === 0.5のときの処理
    case 0.5:
      note.textContent = "半分の分量";
      break;

    // いずれの値とも一致しないときの処理
    default:
      // ➀の文字列を空文字列に設定
      note.textContent = "";
  }
});
```
