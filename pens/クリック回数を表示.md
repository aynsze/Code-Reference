```html
<span id="count-view">0</span>
<button id="count-btn">+ 1 増やす</button>
```

```js
// 1. HTMLのid名「count-view」を取得
const countView = document.getElementById("count-view");
// 2. HTMLのid名「count-btn」を取得
const countBtn = document.getElementById("count-btn");

// 3. 再代入できるように、クリック回数を保持する変数を定義して、初期値を0にする
let count = 0;

// ➁の変数名「countBtn」がクリックされたときに、この関数が実行される
countBtn.addEventListener("click", function () {

  // ➂のクリック回数を保持する変数「count」に1を足す
  count++;      // count = count + 1; と同じ

  // ➀のHTMLのid名「count-view」に、➄で計算したクリック回数を設定
  countView.textContent = count;
});
```
